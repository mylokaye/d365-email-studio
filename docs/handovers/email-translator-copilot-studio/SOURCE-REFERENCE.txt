import { useEffect, useId, useRef, type ReactNode, type RefObject } from 'react';
import { X } from 'lucide-react';
import { Hint } from './Hint';

export function openDialog(ref: RefObject<HTMLDialogElement | null>) {
  const dialog = ref.current;
  if (!dialog || dialog.open) return;
  dialog.returnValue = '';
  dialog.showModal();
  dialog.querySelector<HTMLElement>('[data-dialog-autofocus]')?.focus();
}

function useLightDismiss(ref: RefObject<HTMLDialogElement | null>) {
  useEffect(() => {
    const dialog = ref.current;
    if (!dialog || 'closedBy' in HTMLDialogElement.prototype) return;
    const outside = (e: PointerEvent) => {
      const r = dialog.getBoundingClientRect();
      return e.clientX < r.left || e.clientX > r.right || e.clientY < r.top || e.clientY > r.bottom;
    };
    let pressedOutside = false;
    const down = (e: PointerEvent) => { pressedOutside = outside(e); };
    const up = (e: PointerEvent) => { if (pressedOutside && outside(e)) dialog.close(); pressedOutside = false; };
    dialog.addEventListener('pointerdown', down);
    dialog.addEventListener('pointerup', up);
    return () => { dialog.removeEventListener('pointerdown', down); dialog.removeEventListener('pointerup', up); };
  }, [ref]);
}

export function Dialog({dialogRef, title, description, compact = false, onClose, children}: {
  dialogRef: RefObject<HTMLDialogElement | null>;
  title: string;
  description?: string;
  compact?: boolean;
  onClose?: () => void;
  children: ReactNode;
}) {
  const id = useId();
  useLightDismiss(dialogRef);
  return <dialog ref={dialogRef} className={'paste-dialog' + (compact ? ' confirm-dialog' : '')} {...({closedby: 'any'} as any)} aria-labelledby={id} aria-describedby={description ? `${id}-help` : undefined} onClose={onClose}>
    <div className="paste-heading"><h2 id={id}>{title}</h2><Hint text="Close dialog">{hintId => <button type="button" className="button secondary icon-button dialog-close" aria-label={`Close ${title.replace(/\?$/, '').toLowerCase()} dialog`} aria-describedby={hintId} onClick={() => dialogRef.current?.close()}><X size={20} aria-hidden="true"/></button>}</Hint></div>
    {description && <p id={`${id}-help`}>{description}</p>}
    {children}
  </dialog>;
}

export function ConfirmDialog({open, title, description, confirmLabel, onConfirm, onCancel}: {
  open: boolean; title: string; description: string; confirmLabel: string; onConfirm: () => void; onCancel: () => void;
}) {
  const dialog = useRef<HTMLDialogElement>(null);
  useEffect(() => { if (open) openDialog(dialog); else dialog.current?.close(); }, [open]);
  return <Dialog dialogRef={dialog} title={title} description={description} compact onClose={() => {
    const choice = dialog.current?.returnValue;
    if (dialog.current) dialog.current.returnValue = '';
    if (choice === 'confirm') onConfirm(); else onCancel();
  }}>
    <form method="dialog" className="paste-actions">
      <button className="button secondary" type="submit" value="cancel" data-dialog-autofocus autoFocus>Cancel</button>
      <button className="button primary" type="submit" value="confirm">{confirmLabel}</button>
    </form>
  </Dialog>;
}

import { useEffect, useRef, useState, type ReactNode } from 'react';
import type { Email, Field } from './email';
import { previewDocument } from './preview';

function normalize(text: string) {
  return text.replace(/\s+/g, ' ').trim();
}

function fieldSnippets(field: Field, changes: Record<string, string>, version: 'original' | 'edited'): string[] {
  const edited = version === 'edited' ? changes[field.id] : undefined;
  if (edited !== undefined && edited !== field.original) {
    const parts = field.sections ? edited.replace(/\r\n?/g, '\n').split('\n') : [edited];
    const cleaned = parts.map(normalize).filter(Boolean);
    if (cleaned.length) return cleaned;
  }
  if (field.sections?.length) {
    return field.sections.map(s => normalize(s.original)).filter(Boolean);
  }
  const single = normalize(field.original);
  return single ? [single] : [];
}

function sharedAncestor(a: HTMLElement, b: HTMLElement, root: HTMLElement): HTMLElement {
  let current: HTMLElement | null = a;
  while (current && current !== root) {
    if (current.contains(b)) return current;
    current = current.parentElement;
  }
  return a;
}

function findFieldElement(
  doc: Document,
  email: Email,
  changes: Record<string, string>,
  version: 'original' | 'edited',
  targetId: string
): HTMLElement | null {
  const body = doc.body;
  if (!body) return null;
  const items: { kind: 'text' | 'alt'; text: string; element: HTMLElement }[] = [];
  const walker = doc.createTreeWalker(body, NodeFilter.SHOW_ELEMENT | NodeFilter.SHOW_TEXT);
  let node = walker.nextNode();
  while (node) {
    if (node.nodeType === Node.ELEMENT_NODE) {
      const el = node as HTMLElement;
      if (el.localName === 'img') {
        const alt = normalize(el.getAttribute('alt') ?? '');
        if (alt) items.push({ kind: 'alt', text: alt, element: el });
      }
    } else if (node.nodeType === Node.TEXT_NODE && node.parentElement) {
      const text = normalize(node.nodeValue ?? '');
      if (text) items.push({ kind: 'text', text, element: node.parentElement });
    }
    node = walker.nextNode();
  }

  let cursor = 0;
  for (const field of email.fields) {
    const snippets = fieldSnippets(field, changes, version);
    if (!snippets.length) continue;
    let matchIndex = -1;
    for (let i = cursor; i < items.length; i++) {
      if (items[i].kind === field.kind && items[i].text.includes(snippets[0])) {
        matchIndex = i;
        break;
      }
    }
    if (matchIndex === -1) {
      for (let i = 0; i < cursor; i++) {
        if (items[i].kind === field.kind && items[i].text.includes(snippets[0])) {
          matchIndex = i;
          break;
        }
      }
    }
    if (matchIndex === -1) continue;

    const firstElement = items[matchIndex].element;
    let lastElement = firstElement;
    let nextCursor = items[matchIndex].text === snippets[0] ? matchIndex + 1 : matchIndex;

    if (snippets.length > 1) {
      let searchFrom = matchIndex + 1;
      for (let s = 1; s < snippets.length; s++) {
        for (let i = searchFrom; i < items.length; i++) {
          if (items[i].kind === field.kind && items[i].text.includes(snippets[s])) {
            lastElement = items[i].element;
            searchFrom = i + 1;
            nextCursor = searchFrom;
            break;
          }
        }
      }
    }

    cursor = nextCursor;
    if (field.id === targetId) {
      return firstElement === lastElement ? firstElement : sharedAncestor(firstElement, lastElement, body);
    }
  }
  return null;
}

function scrollPreviewToAsync(element: HTMLElement, behavior: ScrollBehavior, scroller: Window): Promise<void> {
  const bounds = element.getBoundingClientRect();
  // scrollIntoView also scrolls the outer page, taking mobile editors off screen.
  const result = scroller.scrollTo({
    top: scroller.scrollY + bounds.top - Math.max(0, (scroller.innerHeight - bounds.height) / 2),
    left: scroller.scrollX + bounds.left - Math.max(0, (scroller.innerWidth - bounds.width) / 2),
    behavior,
  }) as unknown;
  // Scroll promises originate in the iframe's realm, so instanceof is unreliable.
  if (result && typeof (result as Promise<void>).then === 'function') {
    return Promise.resolve(result as Promise<void>);
  }
  return new Promise(resolve => {
    let timeoutId: ReturnType<typeof setTimeout>;
    const finish = () => {
      clearTimeout(timeoutId);
      scroller.removeEventListener('scrollend', finish);
      scroller.removeEventListener('scroll', onScroll);
      resolve();
    };
    const onScroll = () => {
      clearTimeout(timeoutId);
      timeoutId = setTimeout(finish, 100);
    };
    scroller.addEventListener('scrollend', finish, { once: true });
    scroller.addEventListener('scroll', onScroll, { passive: true });
    timeoutId = setTimeout(finish, 100);
  });
}

export function EmailPreview({
  source,
  original,
  email,
  changes,
  selection,
  actions,
}: {
  source: string;
  original: string;
  email: Email;
  changes: Record<string, string>;
  selection: { id: string; seq: number } | null;
  actions: ReactNode;
}) {
  const [version, setVersion] = useState<'original' | 'edited'>('edited');
  const [display, setDisplay] = useState('');
  const [failure, setFailure] = useState('');
  const iframeRef = useRef<HTMLIFrameElement>(null);
  const canvasRef = useRef<HTMLDivElement>(null);
  const [availableWidth, setAvailableWidth] = useState(620);
  const [emailWidth, setEmailWidth] = useState(620);
  const scale = Math.min(1, availableWidth / emailWidth);
  const savedScrollTop = useRef(0);
  const lastScrolledSeq = useRef(0);
  const scrollRequestId = useRef(0);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const observer = new ResizeObserver(entries => {
      setAvailableWidth(Math.max(1, entries[0].contentRect.width));
    });
    observer.observe(canvas);
    return () => observer.disconnect();
  }, []);

  useEffect(() => {
    let active = true;
    const win = iframeRef.current?.contentWindow;
    if (win) savedScrollTop.current = win.scrollY;
    previewDocument(version === 'edited' ? source : original)
      .then(html => {
        if (active) {
          setDisplay(html);
          setFailure('');
        }
      })
      .catch(() => {
        if (active) {
          setDisplay('');
          setFailure('Preview unavailable. Original HTML is intact.');
        }
      });
    return () => {
      active = false;
    };
  }, [source, original, version]);

  const syncSelection = async (animate: boolean) => {
    const doc = iframeRef.current?.contentDocument;
    const win = iframeRef.current?.contentWindow;
    if (!doc || !win || !selection) return;
    doc.querySelectorAll('[data-preview-active]').forEach(el => el.removeAttribute('data-preview-active'));
    const target = findFieldElement(doc, email, changes, version, selection.id);
    if (!target) return;

    if (!animate) {
      target.setAttribute('data-preview-active', 'true');
      return;
    }

    const requestId = ++scrollRequestId.current;
    const reducedMotion = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches ?? false;
    await scrollPreviewToAsync(target, reducedMotion ? 'auto' : 'smooth', win);
    if (requestId !== scrollRequestId.current) return;
    target.setAttribute('data-preview-active', 'true');
  };

  useEffect(() => {
    if (!selection) return;
    lastScrolledSeq.current = selection.seq;
    void syncSelection(true);
  }, [selection, version]);

  const handleLoad = () => {
    const doc = iframeRef.current?.contentDocument;
    if (doc) setEmailWidth(Math.max(620, doc.documentElement.scrollWidth));
    const win = iframeRef.current?.contentWindow;
    if (win && savedScrollTop.current > 0) {
      win.scrollTo(0, savedScrollTop.current);
    }
    if (selection) {
      const needsScroll = lastScrolledSeq.current !== selection.seq;
      lastScrolledSeq.current = selection.seq;
      void syncSelection(needsScroll);
    }
  };

  return (
    <section className="panel preview-panel" aria-label="Email preview">
      <div className="panel-heading">
        <h2>Preview</h2>
      </div>
      <div className="preview-toolbar">
        <div className="preview-actions">{actions}</div>
        <div className="segmented" role="group" aria-label="Preview version">
          {(['original', 'edited'] as const).map(v => (
            <button key={v} aria-pressed={version === v} className={version === v ? 'active' : ''} onClick={() => setVersion(v)}>
              {v === 'original' ? 'Original' : 'Edited'}
            </button>
          ))}
        </div>
      </div>
      <div className="preview-canvas" ref={canvasRef}>
        {display ? (
          <div className="preview-frame" style={{ width: emailWidth * scale }}>
          <iframe
            ref={iframeRef}
            title="Email display copy"
            sandbox="allow-same-origin"
            referrerPolicy="no-referrer"
            srcDoc={display}
            onLoad={handleLoad}
            style={{ width: emailWidth, height: `${100 / scale}%`, transform: `scale(${scale})` }}
          />
          </div>
        ) : (
          !failure && <p role="status">Loading preview…</p>
        )}
      </div>
      {failure && (
        <div className="error-banner" role="alert">
          {failure}
        </div>
      )}
    </section>
  );
}

import { useEffect, useRef, useState } from 'react';
import { Dialog, openDialog } from './Dialog';

export function HtmlExportDialog({source,onClose}:{source:string|null;onClose:()=>void}) {
  const dialog=useRef<HTMLDialogElement>(null);
  const field=useRef<HTMLTextAreaElement>(null);
  const [status,setStatus]=useState('');
  useEffect(()=>{
    if(source!==null){
      setStatus('');
      openDialog(dialog);
      field.current?.select();
    } else if(dialog.current?.open)dialog.current.close();
  },[source]);
  const copy=async()=>{
    if(source===null)return;
    try{await navigator.clipboard.writeText(source);setStatus('HTML copied.');}
    catch{field.current?.focus();field.current?.select();setStatus('Copy failed. Select the HTML and copy it manually.');}
  };
  return <Dialog dialogRef={dialog} title="Export HTML" onClose={onClose}>
    <label className="visually-hidden" htmlFor="copy-source">Exported HTML</label>
    <textarea ref={field} data-dialog-autofocus autoFocus id="copy-source" name="copy-source" value={source??''} readOnly spellCheck={false}/>
    <p className="export-status" role="status" aria-live="polite">{status}</p>
    <div className="paste-actions"><button className="button primary" type="button" onClick={()=>void copy()}>Copy</button></div>
  </Dialog>;
}

import { useRef, useState } from 'react';
import { ClipboardPaste } from 'lucide-react';
import { MAX_BYTES, looksLikeEmailHtml, readEmail, type Email } from './email';
import { ConfirmDialog, Dialog, openDialog } from './Dialog';

export function PasteHtmlDialog({dirty = false, onLoad}:{dirty?:boolean; onLoad:(email:Email)=>boolean}) {
  const dialog=useRef<HTMLDialogElement>(null);
  const textarea=useRef<HTMLTextAreaElement>(null);
  const [source,setSource]=useState('');
  const [error,setError]=useState('');
  const [pending,setPending]=useState<Email|null>(null);
  const commit=(email:Email)=>{
    if(!onLoad(email))return;
    setPending(null);setSource('');setError('');dialog.current?.close();
  };
  const load=()=>{
    if(!looksLikeEmailHtml(source))return;
    try {
      if(new TextEncoder().encode(source).byteLength>MAX_BYTES) throw new Error('Paste an HTML email smaller than 5 MB.');
      const email=readEmail(source);
      if(dirty){
        setPending(email);
        return;
      }
      commit(email);
    }catch(e){setError(e instanceof Error?e.message:'Could not read this HTML.');textarea.current?.focus();}
  };
  return <>
    <button className="button secondary" type="button" onClick={()=>openDialog(dialog)}><ClipboardPaste size={17} aria-hidden="true"/>Import</button>
    <Dialog dialogRef={dialog} title="Import HTML" onClose={()=>{setPending(null);setError('');}}>
      <form onSubmit={e=>{e.preventDefault();load();}}>

function unavailableReason(t: OnDeviceTranslation, language: string) {
  if (!t.supported) return 'Automatic translation is unavailable in this browser. You can edit the wording manually.';
  if (!t.source) return 'Choose the source language first.';
  if (!language) return 'Choose a language to translate into.';
  if (t.busy) return 'A translation is already in progress.';
  if (t.availability === 'checking') return 'Checking whether this language is available.';
  if (t.availability === 'error') return 'Language check failed. Use Retry to check this language again.';
  if (!t.canTranslate) return 'This language pair is unavailable. Choose another language.';
  return '';
}

function parts(raw: string) {

  for (const m of raw.matchAll(entities)) if (/[\p{C}\p{Z}\s]/u.test(decode(m[0]))) fixed.push({start:m.index!,end:m.index!+m[0].length});
  for (const m of raw.matchAll(/[\p{Cf}\p{Zl}\p{Zp}\u00a0]/gu)) fixed.push({start:m.index!,end:m.index!+m[0].length});

  let end=0;

function groupBlocks(fields:Candidate[]):Field[] {

export function exportEmail(email:Email, changes:Record<string,string>, language?:string) {
  const edits:{start:number;end:number;replacement:string}[]=[];
  for(const field of email.fields) {
    const value=changes[field.id];
    if(value===undefined || value===field.original) continue;
    if(!visible(value)) throw new Error('Translation cannot be empty. Use Reset to restore original.');
    if(/\{\{|\}\}|\{%|%\}|\$\{|\[\[|\]\]|[\p{Cf}\p{Zl}\p{Zp}]/u.test(value)) throw new Error('Enter plain text only (no template tags or invisible characters).');
    const sections=field.sections??[{original:field.original,ranges:field.ranges}];
    const values=field.sections?value.replace(/\r\n?/g,'\n').split('\n'):[value];
    if(values.length!==sections.length) throw new Error(`Keep all ${sections.length} sections on separate lines.`);
    for(let i=0;i<sections.length;i++) {
      if(!visible(values[i])) throw new Error('Each text section needs text. Use Reset to restore original.');
      if(values[i]===sections[i].original)continue;
      for(const range of sections[i].ranges) {
        if(email.source.slice(range.start,range.end)!==range.raw) throw new Error('The original source no longer matches this text block. Reload the email.');
        edits.push({...range,replacement:escapeText(values[i])});
      }
    }
  }
  if(language && email.language && language!==email.language.value) {
    if(!/^[A-Za-z]{2,8}(?:-[A-Za-z0-9]{1,8})*$/.test(language)) throw new Error('Choose a valid language.');
    edits.push({...email.language,replacement:language});
  }
  edits.sort((a,b)=>b.start-a.start);
  let result=email.source;
  for(const edit of edits) result=result.slice(0,edit.start)+edit.replacement+result.slice(edit.end);
  return result;
}

import { exportEmail, type Email, type Field } from './email';
import { francAll } from 'franc-min';
import { iso6393 } from 'iso-639-3';
import { languages } from './languages';
export type Availability='unavailable'|'downloadable'|'downloading'|'available';
export type LanguagePair={sourceLanguage:string;targetLanguage:string};
export type LocalTranslator={translate:(text:string,options?:{signal?:AbortSignal})=>Promise<string>;destroy:()=>void};
export type TranslatorFactory={
  availability:(pair:LanguagePair)=>Promise<Availability>;
  create:(options:LanguagePair & {signal?:AbortSignal;monitor?:(monitor:{addEventListener:(name:'downloadprogress',callback:(event:{loaded:number})=>void)=>void})=>void})=>Promise<LocalTranslator>;
};
export function nativeTranslator():TranslatorFactory|undefined {
  const api=(globalThis as unknown as {Translator?:TranslatorFactory}).Translator;
  return typeof api?.availability==='function' && typeof api?.create==='function'?api:undefined;
}
export function sourceLanguage(email:Email) {
  try{return email.language?.value?new Intl.Locale(email.language.value).language:'';}catch{return '';}
}
export function detectSourceLanguage(email: Email): string {
  const text = email.fields.filter(field => field.kind === 'text').map(field => field.original).join(' ').slice(0, 10000);
  const fallback = sourceLanguage(email);
  if ((text.match(/\p{L}/gu)?.length ?? 0) < 80) return fallback;
  const results = francAll(text);
  if (!results.length || results[0][0] === 'und' || (results[1] && results[0][1] - results[1][1] < 0.1)) return fallback;
  const code = iso6393.find(language => language.iso6393 === results[0][0])?.iso6391;
  return code && languages.some(([supported]) => supported === code) ? code : fallback;
}
function abortError(){return new DOMException('Translation cancelled.','AbortError');}
export function abortable<T>(promise:Promise<T>,signal:AbortSignal):Promise<T> {
  return new Promise((resolve,reject)=>{
    const cancel=()=>reject(abortError());
    if(signal.aborted){promise.catch(()=>{});cancel();return;}
    signal.addEventListener('abort',cancel,{once:true});
    promise.then(value=>{signal.removeEventListener('abort',cancel);signal.aborted?cancel():resolve(value);},error=>{signal.removeEventListener('abort',cancel);reject(error);});
  });
}
export function destroy(translator?:LocalTranslator){try{translator?.destroy();}catch{/* Already closed. */}}
export function openTranslator(api:TranslatorFactory,pair:LanguagePair,signal:AbortSignal,onProgress:(value:number)=>void) {
  if(signal.aborted)return Promise.reject(abortError());
  // Called directly from the click handler, before any await, to retain activation.
  const creating=api.create({...pair,signal,monitor(monitor){monitor.addEventListener('downloadprogress',event=>{
    if(!signal.aborted && Number.isFinite(event.loaded))onProgress(Math.max(0,Math.min(1,event.loaded)));
  });}});
  void creating.then(session=>{if(signal.aborted)destroy(session);},()=>{});
  return abortable(creating,signal);
}
export async function translateField(session:LocalTranslator,email:Email,field:Field,signal:AbortSignal) {
  const sections=field.sections??[{original:field.original}];
  const result:string[]=[];
  for(const section of sections) {
    if(signal.aborted)throw abortError();
    // Translate only editable wording, not source markup, destinations or protected expressions.
    const translated=await abortable(session.translate(section.original,{signal}),signal);
    if(typeof translated!=='string')throw new Error('Chrome returned an invalid translation.');
    result.push(translated.trim().replace(/\s*\r?\n\s*/g,' '));
  }
  const value=result.join(field.sections?'\n':'');
  exportEmail(email,{[field.id]:value});
  return value;
}

// Preview HTML is a disposable, sanitized display copy. It is never exported.
export async function previewDocument(source:string) {
  const native=(Document as unknown as NativeParser).parseHTML;
  let document:HTMLElement;
  if(native) {
    document=native.call(Document,source,{sanitizer:{removeElements:blockedTags,removeAttributes:blockedAttrs}}).documentElement;
  } else {
    // Safari and older browsers receive the same isolated display-copy behavior.
    const {default:DOMPurify}=await import('dompurify');
    document=DOMPurify.sanitize(source,{
      WHOLE_DOCUMENT:true,RETURN_DOM:true,ALLOW_DATA_ATTR:false,
      FORBID_TAGS:blockedTags,FORBID_ATTR:blockedAttrs,
    }) as HTMLElement;
  }
  for(const el of document.querySelectorAll('[src]')) {
    const value=el.getAttribute('src')??'';
    const permitted=el.localName==='img' && (/^data:image\/(?:png|jpeg|gif|webp);/i.test(value) || /^https:\/\//i.test(value));
    if(!permitted) {el.removeAttribute('src');el.setAttribute('title','Remote image hidden in preview');}
    else el.setAttribute('referrerpolicy','no-referrer');
  }
  const csp=`default-src 'none'; script-src 'none'; connect-src 'none'; style-src 'unsafe-inline'; img-src data: https:; font-src 'none'; frame-src 'none'; form-action 'none'; base-uri 'none';`;
  return `<!doctype html><html><head><meta http-equiv="Content-Security-Policy" content="${csp}"><meta name="referrer" content="no-referrer"><meta name="viewport" content="width=device-width,initial-scale=1"><style>body{margin:0;overflow-wrap:anywhere}:where(th.inner[role="presentation"]:not([align])){text-align:start}img{max-width:100%}a{cursor:default}[data-preview-active]{outline:2px solid #174f37!important;outline-offset:3px!important;border-radius:3px}</style>${document.querySelector('head')?.innerHTML??''}</head>${document.querySelector('body')?.outerHTML??'<body></body>'}</html>`;
}

import {describe, expect, it} from 'vitest';
import {draftReducer, emptyDrafts, hasDraftChanges} from './translation-drafts';

describe('language drafts', () => {
  it('keeps translated wording separate when selecting another target and restores it on return', () => {
    let state = draftReducer(emptyDrafts, {type: 'select', language: 'fr'});
    state = draftReducer(state, {type: 'change', id: 'title', value: 'Bonjour'});
    state = draftReducer(state, {type: 'select', language: 'de'});
    expect(state.changes).toEqual({});
    expect(hasDraftChanges(state, [{id: 'title', original: 'Hello'}])).toBe(true);
    state = draftReducer(state, {type: 'change', id: 'title', value: 'Hallo'});
    state = draftReducer(state, {type: 'select', language: 'fr'});
    expect(state.changes.title).toBe('Bonjour');
    state = draftReducer(state, {type: 'select', language: 'de'});
    expect(state.changes.title).toBe('Hallo');
  });
  it('assigns pre-selection manual edits to the first target without duplicating a draft', () => {
    let state = draftReducer(emptyDrafts, {type: 'change', id: 'title', value: 'Bonjour'});
    state = draftReducer(state, {type: 'select', language: 'fr'});
    expect(state.changes.title).toBe('Bonjour');
    expect(state.drafts).toEqual({});
  });
  it('does not overwrite newer manual edits with an older translation result', () => {
    const state = draftReducer(emptyDrafts, {type: 'change', id: 'title', value: 'My wording'});
    expect(draftReducer(state, {type: 'apply', id: 'title', value: 'Bonjour', expected: undefined})).toBe(state);
  });
  it('clears every language draft on reset', () => {
    let state = draftReducer(emptyDrafts, {type: 'select', language: 'fr'});
    state = draftReducer(state, {type: 'change', id: 'title', value: 'Bonjour'});
    state = draftReducer(state, {type: 'select', language: 'de'});
    state = draftReducer(state, {type: 'reset'});
    expect(state).toEqual(emptyDrafts);
    expect(draftReducer(state, {type: 'select', language: 'fr'}).changes).toEqual({});
  });
});

export type Changes = Record<string, string>;
export type DraftState = {language: string; changes: Changes; drafts: Record<string, Changes>};
export const emptyDrafts: DraftState = {language: '', changes: {}, drafts: {}};
type Action = {type: 'select'; language: string} | {type: 'change'; id: string; value: string} | {type: 'apply'; id: string; value: string; expected: string | undefined} | {type: 'reset'};

export function draftReducer(state: DraftState, action: Action): DraftState {
  if (action.type === 'reset') return emptyDrafts;
  if (action.type === 'select') {
    if (state.language === action.language) return state;
    const drafts = {...state.drafts};
    if (state.language) drafts[state.language] = state.changes;
    // Edits entered before choosing a target belong to the first selected language.
    const changes = drafts[action.language] ?? (state.language ? {} : state.changes);
    return {language: action.language, changes, drafts};
  }
  if (action.type === 'apply' && state.changes[action.id] !== action.expected) return state;
  return {...state, changes: {...state.changes, [action.id]: action.value}};
}

export function hasDraftChanges(state: DraftState, fields: {id: string; original: string}[]) {
  const changed = (changes: Changes) => fields.some(field => changes[field.id] !== undefined && changes[field.id] !== field.original);
  return changed(state.changes) || Object.entries(state.drafts).some(([language, changes]) => language !== state.language && changed(changes));
}

      for(let i=0;i<fields.length;i++) {
        const field=fields[i];
        if(signal.aborted || latest.current.email!==initial.email || latest.current.target!==initial.target)return;
        if(latest.current.changes[field.id]!==initial.changes[field.id]){skipped++;continue;}
        setStatus(`Translating ${i+1} of ${fields.length}…`);
        const value=await translateField(current.session,initial.email,field,signal);
        if(signal.aborted || latest.current.email!==initial.email || latest.current.target!==initial.target)return;
        if(latest.current.changes[field.id]!==initial.changes[field.id]){skipped++;continue;}
        latest.current.apply(field.id,value,initial.changes[field.id]);completed++;
      }
      setStatus(skipped ? `${skipped} manually edited block${skipped===1?' was':'s were'} kept.` : '');
      setState({key:pairKey,value:'available'});
    }catch(error){
      if(!signal.aborted && job.current===current){
        const reason=error instanceof Error?error.message:'Translation failed.';
        setStatus(`Translation stopped (${completed} completed). ${reason}`);
      }
    }finally{
      destroy(current.session);
      if(job.current===current){job.current=undefined;setBusy(false);setProgress(null);}
    }
  };
  return {supported:!!api,source,setSource,detectedSource,availability,canTranslate,busy,status,progress,untouched,run,cancel,retry:()=>setCheckVersion(version=>version+1)};
}
